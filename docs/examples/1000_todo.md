# Persistent Todo List

This example extends the [`Data Model` and `Effects` Counter](500_effect_counter.html)
with browser persistence. Todo items belong to the global `Data Model`. The `FSM` keeps command data in its `Context`
and copies it into the `Event Meta` that its `Effects` consume.

## `Data Model`

```ts
interface ITodoItem {
	id: string;
	text: string;
}

interface ITodoModel {
	items: Record<string, ITodoItem>;
}

type TTodoSnapshot = Pick<ITodoModel, 'items'>;

const defaultModel: ITodoModel = { items: {} };
```

The `Data Model` is a plain snapshot. `ModelStore` owns it in memory, while the
`Storage` binding below persists only the `items` projection.

## `FSM` and `Effects`

```mermaid

  READY --> ADDED: ADD
  READY --> REMOVED: REMOVE
  READY --> CLEARED: CLEAR
  ADDED --> ADDED: ADD

  ADDED --> CLEARED: CLEAR
  REMOVED --> ADDED: ADD
  REMOVED --> REMOVED: REMOVE
  REMOVED --> CLEARED: CLEAR
  CLEARED --> ADDED: ADD
  CLEARED --> REMOVED: REMOVE
  CLEARED --> CLEARED: CLEAR

  note right of READY
    +Init
    subscribe/addTodo ADD ($id, $text) <= ($id, $text)
    subscribe/removeTodo REMOVE ($id) <= ($id)
    subscribe/clearTodos CLEAR
  end note

  note right of ADDED
    #{id, text} <= $id, $text
    effect/todoAdded <= (id, text)
  end note

  note right of REMOVED
    #{id} <= $id
    effect/todoRemoved <= (id)
  end note

  note right of CLEARED
    effect/todosCleared
  end note
```

Generate a TypeScript `FSM` named `TodoFlow`. The generated `Slice` factory
requires one `Model Transformer` for each declared `Effect`.

Pass `true` as the store factory's second argument during development to freeze
snapshots and validate their contents.

```ts
import {
	CoreLoop,
	createModelFunctions,
	EffectScheduler,
	hydrateModel,
	ModelStore,
	removeModelProperty,
	StorageSyncLoop,
	type TStorageBinding,
} from '@yantrix/core';
import { LocalStorageAdapter } from '@yantrix/storage';
import {
	createTodoFlowSlice,
	eventDictionary,
	type TTodoFlowEffectEvent,
	type TTodoFlowEventMeta,
} from './generated/TodoFlow.js';

const todoStorage = new LocalStorageAdapter<TTodoSnapshot>({
	id: 'todo-list',
	key: 'yantrix:todo-list:v1',
});

const todoBinding: TStorageBinding<ITodoModel, TTodoSnapshot> = {
	storage: todoStorage,
	select: model => ({ items: model.items }),
	hydrate: (model, snapshot) => ({ ...model, items: snapshot.items }),
};

// Hydrate before constructing the store and the `Main Loop`.
const hydration = await hydrateModel(defaultModel, [todoBinding]);
for (const failure of hydration.failures) {
	console.error('Todo hydration failed', failure);
}

function createTodoStore(initial: ITodoModel, development = false) {
	return new ModelStore(initial, {
		development: { freeze: development, validateSerializable: development },
	});
}

const modelStore = createTodoStore(hydration.model);
const fx = createModelFunctions<ITodoModel, TTodoFlowEffectEvent, TTodoFlowEventMeta>();
const added = fx.forEvent<typeof eventDictionary.todoAdded>();
const removed = fx.forEvent<typeof eventDictionary.todoRemoved>();
const todoSlice = createTodoFlowSlice<ITodoModel>({
	todoAdded: added.transformer(
		model => model.items,
		(model, items) => ({ ...model, items }),
		(event, items) => {
			const id = event.meta?.id;
			const text = event.meta?.text;
			if (typeof id !== 'string' || !id || typeof text !== 'string' || !text) return items;
			return { ...items, [id]: { id, text } };
		},
	),
	todoRemoved: removed.transformer(
		model => model.items,
		(model, items) => ({ ...model, items }),
		(event, items) => {
			const id = event.meta?.id;
			return typeof id === 'string' ? removeModelProperty(items, id) : items;
		},
	),
	todosCleared: (_event, model) => Object.keys(model.items).length === 0
		? model
		: { ...model, items: {} },
});

const effectScheduler = new EffectScheduler({ store: modelStore });
const coreLoop = new CoreLoop({ effectScheduler });
coreLoop.registerSlice(todoSlice).start();

const storageSync = new StorageSyncLoop({
	store: modelStore,
	bindings: [todoBinding],
	session: hydration.session,
	debounceMs: 150,
	onError: failure => console.error('Todo persistence failed', failure),
}).start();

// Persist defaults or a successfully migrated snapshot even before a command.
storageSync.requestSync();
```

`LocalStorageAdapter` can accept an injected `storage`, `serialize` and
`deserialize`, which makes the same composition testable without browser
globals.

## UI commands and rendering

The UI publishes commands and renders committed `Data Model` snapshots. It never
edits the `FSM` `Context` or `Storage` directly.

```ts
function render(model: ITodoModel): void {
	// Replace this with React, DOM, CLI or an API `Destination`.
	console.table(Object.values(model.items));
}

render(modelStore.get());
const stopRendering = modelStore.subscribe(render);

let acceptingCommands = true;

interface ITodoCommandMeta {
	addTodo: ITodoItem;
	removeTodo: Pick<ITodoItem, 'id'>;
	clearTodos: null;
}

async function dispatchCommand<Name extends keyof ITodoCommandMeta>(
	name: Name,
	meta: ITodoCommandMeta[NoInfer<Name>],
): Promise<void> {
	if (!acceptingCommands) throw new Error('Todo list is shutting down');
	coreLoop.getBus().dispatch({ event: eventDictionary[name], meta });
	await coreLoop.whenIdle();
}

export function addTodo(id: string, text: string): Promise<void> {
	return dispatchCommand('addTodo', { id, text });
}

export function removeTodo(id: string): Promise<void> {
	return dispatchCommand('removeTodo', { id });
}

export function clearTodos(): Promise<void> {
	return dispatchCommand('clearTodos', null);
}

async function shutdown(): Promise<void> {
	acceptingCommands = false;
	try {
		await coreLoop.whenIdle();
	} finally {
		stopRendering();
		coreLoop.unregisterSlice(todoSlice.id);
		coreLoop.stop();
		await storageSync.stop(true);
	}
}
```

`coreLoop.whenIdle()` waits for the command's `Event` cascade and `Effect` batch. Persistence is
asynchronous. During shutdown, stop accepting commands and drain the running `Main Loop` before
disconnecting the `Slice`. Flush the `Sync Loop` to persist the final committed snapshot.

```ts
await addTodo('first', 'Try Yantrix Effects');
await storageSync.whenIdle();
```

Reloading the application repeats `hydrateModel()` and restores the saved
`items` projection before the UI and `Main Loop` start.
